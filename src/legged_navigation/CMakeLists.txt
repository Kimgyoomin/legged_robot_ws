cmake_minimum_required(VERSION 3.16)
project(legged_navigation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(rubi_heightmap_step_wavefront_planner REQUIRED)
add_library(pure_pursuit_core src/pure_pursuit_core.cpp)
target_compile_features(pure_pursuit_core PUBLIC cxx_std_17)
target_include_directories(pure_pursuit_core PUBLIC include)
add_executable(pure_pursuit src/simple_pure_pursuit_controller.cpp)
target_link_libraries(pure_pursuit pure_pursuit_core)
ament_target_dependencies(pure_pursuit rclcpp std_msgs nav_msgs geometry_msgs visualization_msgs tf2_ros tf2_geometry_msgs)
add_executable(obstacle_cloud src/obstacle_cloud.cpp)
ament_target_dependencies(obstacle_cloud rclcpp sensor_msgs std_msgs tf2_ros tf2_msgs)
add_library(sector_filter_core STATIC src/sector_filter.cpp)
target_compile_features(sector_filter_core PUBLIC cxx_std_17)
target_include_directories(sector_filter_core PUBLIC include)
target_link_libraries(sector_filter_core Eigen3::Eigen)
ament_target_dependencies(sector_filter_core sensor_msgs)
add_executable(costmap_sector_filter src/costmap_sector_filter.cpp)
target_link_libraries(costmap_sector_filter sector_filter_core)
ament_target_dependencies(costmap_sector_filter rclcpp rcl_interfaces sensor_msgs geometry_msgs tf2_ros)
add_executable(fastlio_velocity_feedback src/fastlio_velocity_feedback.cpp)
target_compile_features(fastlio_velocity_feedback PUBLIC cxx_std_17)
target_include_directories(fastlio_velocity_feedback PRIVATE include)
target_link_libraries(fastlio_velocity_feedback Eigen3::Eigen)
ament_target_dependencies(fastlio_velocity_feedback rclcpp rcl_interfaces nav_msgs tf2_ros)
install(TARGETS costmap_sector_filter fastlio_velocity_feedback DESTINATION lib/${PROJECT_NAME})
add_library(height_planner SHARED src/height_planner.cpp)
target_compile_features(height_planner PUBLIC cxx_std_17)
ament_target_dependencies(height_planner rclcpp rcl_interfaces nav2_core nav2_costmap_2d nav2_util pluginlib
  nav_msgs sensor_msgs geometry_msgs tf2_ros tf2_geometry_msgs rubi_heightmap_step_wavefront_planner)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS pure_pursuit obstacle_cloud height_planner ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  foreach(test_name sector_filter height_filter velocity_feedback)
    ament_add_gtest(test_${test_name} test/test_${test_name}.cpp)
    target_link_libraries(test_${test_name} sector_filter_core Eigen3::Eigen)
    ament_target_dependencies(test_${test_name} sensor_msgs nav_msgs)
  endforeach()
  ament_add_gtest(test_pure_pursuit_core test/test_pure_pursuit_core.cpp)
  target_link_libraries(test_pure_pursuit_core pure_pursuit_core)
endif()
ament_package()
