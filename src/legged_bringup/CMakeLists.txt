cmake_minimum_required(VERSION 3.16)
project(legged_bringup)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/frozen_map scripts/command_watchdog DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(delivery_contract test/test_delivery.py ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
  ament_add_pytest_test(costmap_guard test/test_costmap_guard.py ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(behaviortree_cpp_v3 REQUIRED)
  ament_add_gtest(navigation_recovery test/test_navigation_recovery.cpp)
  ament_target_dependencies(navigation_recovery behaviortree_cpp_v3)
  target_compile_definitions(navigation_recovery PRIVATE
    NAVIGATION_CONFIG_DIR="${CMAKE_CURRENT_SOURCE_DIR}/config")
endif()
ament_package()
